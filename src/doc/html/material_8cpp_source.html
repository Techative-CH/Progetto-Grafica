157void Eng::Material::render()
158{
159 // Enable and bind the texture only if the material has one
160 if (texture != nullptr)
161 {
162 glEnable(GL_TEXTURE_2D);
163 texture->render();
164 }
165 else
166 {
167 glDisable(GL_TEXTURE_2D);
168 }
169
170 glMaterialfv(GL_FRONT_AND_BACK, GL_EMISSION, glm::value_ptr(emission)); // Emitted light
171 glMaterialfv(GL_FRONT_AND_BACK, GL_AMBIENT, glm::value_ptr(ambient)); // Ambient light reflection
172 glMaterialfv(GL_FRONT_AND_BACK, GL_DIFFUSE, glm::value_ptr(diffuse)); // Diffuse light reflection
173 glMaterialfv(GL_FRONT_AND_BACK, GL_SPECULAR, glm::value_ptr(specular)); // Specular light reflection
174 glMaterialf(GL_FRONT_AND_BACK, GL_SHININESS, shininess); // Specular highlight size
175}
