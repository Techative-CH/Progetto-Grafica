412void ENG_API Eng::Base::render(Eng::List* list)
413{
414 if (!list || !currentCamera)
415 return;
416
417 // Load the projection matrix of the active camera
418 glm::mat4 projectionMatrix = currentCamera->getProjectionMatrix();
419
420 glMatrixMode(GL_PROJECTION);
421 glLoadMatrixf(glm::value_ptr(projectionMatrix));
422
423 // Get the view matrix of the active camera
424 glm::mat4 viewMatrix = currentCamera->getViewMatrix();
425
426 // Get the maximum number of lights supported by OpenGL
427 GLint maxLights = 0;
428 glGetIntegerv(GL_MAX_LIGHTS, &maxLights);
429
430 GLint lightIndex = 0;
431
432 // Render each element in the render list
433 for (const RenderElement& element : list->getElements())
434 {
435 if (element.node == nullptr)
436 continue;
437
438 // Configure light nodes
439 if (auto* light = dynamic_cast<Light*>(element.node))
440 {
441 if (lightIndex >= maxLights)
442 continue;
443
444 glEnable(GL_LIGHTING);
445
446 // Assign an OpenGL light identifier
447 GLenum lightId = GL_LIGHT0 + lightIndex;
448 glEnable(lightId);
449
450 light->renderLight(element.worldMatrix, viewMatrix, lightId);
451
452 ++lightIndex;
453 }
454 else
455 {
456 // Transform the object from local space to camera space
457 glm::mat4 modelViewMatrix = viewMatrix * element.worldMatrix;
458
459 glMatrixMode(GL_MODELVIEW);
460 glLoadMatrixf(glm::value_ptr(modelViewMatrix));
461
462 element.node->render();
463 }
464 }
465
466 // Enable lighting only if at least one light was rendered
467 if (lightIndex > 0)
468 glEnable(GL_LIGHTING);
469 else
470 glDisable(GL_LIGHTING);
471}

529void Eng::Base::calculateFPS()
530{
531 ++frameCount;
532
533 auto now = std::chrono::steady_clock::now();
534 float elapsed = std::chrono::duration<float>(now - lastFpsTime).count();
535
536 // Update the FPS value once per second
537 if (elapsed >= 1.0f)
538 {
539 fps = static_cast<float>(frameCount) / elapsed;
540 frameCount = 0;
541 lastFpsTime = now;
542 }

554void Eng::Base::renderText(const std::string& text, int x, int y, float r, float g, float b)
555{
556 int width = glutGet(GLUT_WINDOW_WIDTH);
557 int height = glutGet(GLUT_WINDOW_HEIGHT);
558
559 // Save the OpenGL states modified by the overlay
560 GLboolean lightingEnabled = glIsEnabled(GL_LIGHTING);
561 GLboolean textureEnabled = glIsEnabled(GL_TEXTURE_2D);
562 GLboolean depthEnabled = glIsEnabled(GL_DEPTH_TEST);
563
564 // Save the current rendering color
565 GLfloat previousColor[4];
566 glGetFloatv(GL_CURRENT_COLOR, previousColor);
567
568 // Save the currently active matrix mode
569 GLint previousMatrixMode;
570 glGetIntegerv(GL_MATRIX_MODE, &previousMatrixMode);
571
572 // Overlay must not be affected by the 3D scene
573 glDisable(GL_LIGHTING);
574 glDisable(GL_TEXTURE_2D);
575 glDisable(GL_DEPTH_TEST);
576
577 // Create a 2D orthographic projection matching the window size
578 glm::mat4 projection = glm::ortho(
579 0.0f,
580 static_cast<float>(width),
581 0.0f,
582 static_cast<float>(height),
583 -1.0f,
584 1.0f
585 );
586
587 glMatrixMode(GL_PROJECTION);
588 glLoadMatrixf(glm::value_ptr(projection));
589
590 // Text does not require camera or world transformations
591 glm::mat4 modelView(1.0f);
592
593 glMatrixMode(GL_MODELVIEW);
594 glLoadMatrixf(glm::value_ptr(modelView));
595
596 // Configure text color and screen position
597 glColor3f(r, g, b);
598 glRasterPos2i(x, y);
599
600 // Render each character using the FreeGLUT bitmap font
601 for (char character : text)
602 {
603 glutBitmapCharacter(
604 GLUT_BITMAP_8_BY_13,
605 character
606 );
607 }
608
609 // Restore the previous color
610 glColor4fv(previousColor);
611
612 // Restore lighting state
613 if (lightingEnabled)
614 glEnable(GL_LIGHTING);
615 else
616 glDisable(GL_LIGHTING);
617
618 // Restore texture state
619 if (textureEnabled)
620 glEnable(GL_TEXTURE_2D);
621 else
622 glDisable(GL_TEXTURE_2D);
623
624 // Restore depth testing state
625 if (depthEnabled)
626 glEnable(GL_DEPTH_TEST);
627 else
628 glDisable(GL_DEPTH_TEST);
629
630 // Restore the active camera matrices
631 if (currentCamera != nullptr)
632 {
633 glm::mat4 projectionMatrix = currentCamera->getProjectionMatrix();
634 glm::mat4 viewMatrix = currentCamera->getViewMatrix();
635
636 glMatrixMode(GL_PROJECTION);
637 glLoadMatrixf(glm::value_ptr(projectionMatrix));
638
639 glMatrixMode(GL_MODELVIEW);
640 glLoadMatrixf(glm::value_ptr(viewMatrix));
641 }
642
643 // Restore the previously active matrix mode
644 glMatrixMode(previousMatrixMode);
645}
